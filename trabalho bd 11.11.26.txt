CREATE DATABASE IF NOT EXISTS livraria_db;

USE livraria_db;

CREATE TABLE tb_secao (
    id_secao INT AUTO_INCREMENT,
    nome VARCHAR(60) NOT NULL,
    descricao VARCHAR(255) NOT NULL,
    CONSTRAINT PK_tb_secao PRIMARY KEY (id_secao)
);

CREATE TABLE tb_categoria (
    id_categoria INT AUTO_INCREMENT,
    nome VARCHAR(80) NOT NULL,
    descricao VARCHAR(255) NOT NULL,
    CONSTRAINT PK_tb_categoria PRIMARY KEY (id_categoria)
);

CREATE TABLE tb_editora (
    cnpj CHAR(14),
    nome VARCHAR(100) NOT NULL,
    telefone VARCHAR(15),
    email VARCHAR(100),
    CONSTRAINT PK_tb_editora PRIMARY KEY (cnpj)
);

CREATE TABLE tb_autor (
    cpf CHAR(11),
    nome VARCHAR(100) NOT NULL,
    telefone VARCHAR(15),
    email VARCHAR(100),
    CONSTRAINT PK_tb_autor PRIMARY KEY (cpf)
);

CREATE TABLE tb_fornecedor (
    cnpj CHAR(14),
    razao_social VARCHAR(120) NOT NULL,
    telefone VARCHAR(15),
    email VARCHAR(100),
    endereco VARCHAR(200),
    CONSTRAINT PK_tb_fornecedor PRIMARY KEY (cnpj)
);

CREATE TABLE tb_funcionario (
    id_funcionario INT AUTO_INCREMENT,
    nome VARCHAR(100) NOT NULL,
    telefone VARCHAR(15),
    cpf CHAR(11) UNIQUE,
    CONSTRAINT PK_tb_funcionario PRIMARY KEY (id_funcionario)
);

CREATE TABLE tb_prateleira (
    id_prateleira INT AUTO_INCREMENT,
    nome VARCHAR(60) NOT NULL,
    id_secao INT NOT NULL,
    CONSTRAINT PK_tb_prateleira PRIMARY KEY (id_prateleira),
    CONSTRAINT FK_tb_prateleira_tb_secao FOREIGN KEY (id_secao) 
        REFERENCES tb_secao (id_secao)
        ON DELETE RESTRICT ON UPDATE CASCADE
);

CREATE TABLE tb_cliente (
    cpf CHAR(11),
    nome VARCHAR(100) NOT NULL,
    telefone VARCHAR(15),
    id_funcionario INT NOT NULL,
    CONSTRAINT PK_tb_cliente PRIMARY KEY (cpf),
    CONSTRAINT FK_tb_cliente_tb_funcionario FOREIGN KEY (id_funcionario) 
        REFERENCES tb_funcionario (id_funcionario)
        ON DELETE RESTRICT ON UPDATE CASCADE
);

CREATE TABLE tb_livro (
    isbn CHAR(13),
    nome VARCHAR(150) NOT NULL,
    preco DECIMAL(10,2) NOT NULL,
    quantidade INT NOT NULL DEFAULT 0,
    id_categoria INT NOT NULL,
    cnpj_editora CHAR(14) NOT NULL,
    id_prateleira INT NOT NULL,
    CONSTRAINT PK_tb_livro PRIMARY KEY (isbn),
    CONSTRAINT FK_tb_livro_tb_categoria FOREIGN KEY (id_categoria) 
        REFERENCES tb_categoria (id_categoria)
        ON DELETE RESTRICT ON UPDATE CASCADE,
    CONSTRAINT FK_tb_livro_tb_editora FOREIGN KEY (cnpj_editora) 
        REFERENCES tb_editora (cnpj)
        ON DELETE RESTRICT ON UPDATE CASCADE,
    CONSTRAINT FK_tb_livro_tb_prateleira FOREIGN KEY (id_prateleira) 
        REFERENCES tb_prateleira (id_prateleira)
        ON DELETE RESTRICT ON UPDATE CASCADE
);

CREATE TABLE tb_compra (
    id_compra INT AUTO_INCREMENT,
    data_compra DATE NOT NULL,
    forma_pagamento VARCHAR(30) NOT NULL,
    valor_total DECIMAL(10,2) NOT NULL,
    cpf_cliente CHAR(11) NOT NULL,
    id_funcionario INT NOT NULL,
    CONSTRAINT PK_tb_compra PRIMARY KEY (id_compra),
    CONSTRAINT FK_tb_compra_tb_cliente FOREIGN KEY (cpf_cliente) 
        REFERENCES tb_cliente (cpf)
        ON DELETE RESTRICT ON UPDATE CASCADE,
    CONSTRAINT FK_tb_compra_tb_funcionario FOREIGN KEY (id_funcionario) 
        REFERENCES tb_funcionario (id_funcionario)
        ON DELETE RESTRICT ON UPDATE CASCADE
);

CREATE TABLE tb_livro_autor (
    isbn CHAR(13),
    cpf_autor CHAR(11),
    CONSTRAINT PK_tb_livro_autor PRIMARY KEY (isbn, cpf_autor),
    CONSTRAINT FK_tb_livro_autor_tb_livro FOREIGN KEY (isbn) 
        REFERENCES tb_livro (isbn)
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT FK_tb_livro_autor_tb_autor FOREIGN KEY (cpf_autor) 
        REFERENCES tb_autor (cpf)
        ON DELETE RESTRICT ON UPDATE CASCADE
);

CREATE TABLE tb_livro_fornecedor (
    isbn CHAR(13),
    cnpj_fornecedor CHAR(14),
    CONSTRAINT PK_tb_livro_fornecedor PRIMARY KEY (isbn, cnpj_fornecedor),
    CONSTRAINT FK_tb_livro_fornecedor_tb_livro FOREIGN KEY (isbn) 
        REFERENCES tb_livro (isbn)
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT FK_tb_livro_fornecedor_tb_fornecedor FOREIGN KEY (cnpj_fornecedor) 
        REFERENCES tb_fornecedor (cnpj)
        ON DELETE RESTRICT ON UPDATE CASCADE
);

CREATE TABLE tb_item_compra (
    id_compra INT,
    isbn CHAR(13),
    quantidade INT NOT NULL,
    valor_unitario DECIMAL(10,2) NOT NULL,
    CONSTRAINT PK_tb_item_compra PRIMARY KEY (id_compra, isbn),
    CONSTRAINT FK_tb_item_compra_tb_compra FOREIGN KEY (id_compra) 
        REFERENCES tb_compra (id_compra)
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT FK_tb_item_compra_tb_livro FOREIGN KEY (isbn) 
        REFERENCES tb_livro (isbn)
        ON DELETE RESTRICT ON UPDATE CASCADE
);
